cmake_minimum_required(VERSION 3.16)
project(marl_racing LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(AGILICIOUS_DIR "" CACHE PATH "Patched agilicious_internal checkout")
if(NOT EIGEN_INCLUDE_DIR)
  find_package(Eigen3 3.4 REQUIRED NO_MODULE)
  set(EIGEN_INCLUDE_DIR "${EIGEN3_INCLUDE_DIR}")
endif()
find_package(OpenMP REQUIRED)
if(PYBIND11_SOURCE_DIR)
  add_subdirectory("${PYBIND11_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/pybind11")
else()
  find_package(pybind11 CONFIG REQUIRED)
endif()
include(native/agilib_sources.cmake)
file(GLOB_RECURSE RACING_SOURCES CONFIGURE_DEPENDS native/src/*.cpp)
pybind11_add_module(flightgym native/bindings.cpp ${RACING_SOURCES} ${AGILIB_SOURCES})
target_include_directories(flightgym PRIVATE "${EIGEN_INCLUDE_DIR}" native/include "${AGILICIOUS_DIR}/agilib/include")
target_link_libraries(flightgym PRIVATE OpenMP::OpenMP_CXX)
target_compile_definitions(flightgym PRIVATE EIGEN_STACK_ALLOCATION_LIMIT=1048576)
target_compile_options(flightgym PRIVATE -O3 -fno-finite-math-only)
set_target_properties(flightgym PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python")
